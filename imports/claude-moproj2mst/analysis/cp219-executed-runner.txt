function inventory(root) {
  const rootStat=fs.lstatSync(root);
  if(rootStat.isSymbolicLink() || !rootStat.isDirectory()) fail(`unsafe manifest root: ${root}`);
  const entries=[];
  function walk(dir, rel='') {
    for (const d of fs.readdirSync(dir,{withFileTypes:true})) {
      const r=rel ? `${rel}/${d.name}` : d.name, p=path.join(dir,d.name);
      if (d.isSymbolicLink()) fail(`link/reparse entry rejected: ${p}`);
      if (d.isDirectory()) { entries.push({kind:'directory',path:r}); walk(p,r); }
      else if (d.isFile()) entries.push({kind:'file',path:r,bytes:String(fs.statSync(p).size),sha256:digest(p)});
      else fail(`unsupported entry: ${p}`);
    }
  }
  walk(root); entries.sort((a,b)=>a.path.localeCompare(b.path,'en'));
  return {root:path.resolve(root),entries};
}
function encodeQChar(s) {
  const units=[]; for(let i=0;i<s.length;i++) units.push(s.charCodeAt(i).toString(16).padStart(4,'0'));
  const encoded=units.join('');
  if (!/^(?:[0-9a-f]{4})+$/.test(encoded) || encoded.length!==4*s.length) fail('invalid QChar encoding');
  const decoded=units.map(x=>String.fromCharCode(parseInt(x,16))).join('');
  if(decoded!==s) fail('QChar round-trip mismatch');
  return encoded;
}
function patchIds(bytes) {
  const before=Buffer.from(bytes), matches=[...before.toString('latin1').matchAll(/<Operation>[^]*?<Id>([12])<\/Id>[^]*?<Type>/g)];
  // Use the anchored direct operation-ID fragments; require exactly IDs 1 then 2.
  const text=before.toString('latin1');
  const re=/<Operation>(?:(?!<Operation>).)*?<Id>([12])<\/Id>(?:(?!<Operation>).)*?<Type>/gs;
  const found=[...text.matchAll(re)];
  if(found.length!==2 || found.map(x=>x[1]).join(',')!=='1,2') fail('expected exactly two direct operation IDs 1,2');
  let out=Buffer.from(before); const offsets=[];
  for(let i=found.length-1;i>=0;i--) {
    const m=found[i], token=`<Id>${m[1]}</Id>`, start=m.index+m[0].indexOf(token)+4;
    offsets.unshift(start); out=Buffer.concat([out.subarray(0,start),Buffer.from('-'),out.subarray(start)]);
  }
  // Reconstruct from the source to prove there are exactly two inserted minus bytes.
  if(out.length!==before.length+2) fail('mutation length is not exactly two inserted bytes');
  let reconstructed=Buffer.from(before);
  for(const at of [...offsets].reverse()) reconstructed=Buffer.concat([reconstructed.subarray(0,at),Buffer.from('-'),reconstructed.subarray(at)]);
  if(!out.equals(reconstructed)) fail('variant has a byte difference beyond the two required minus insertions');
  return {bytes:out,offsets};
}
function parseEvent(line) {
  // FILETIME is an unsigned 64-bit value; retain its raw decimal text across JSON parsing.
  const event=JSON.parse(line.replace(/("creation_filetime"\s*:\s*)(\d+)/g,'$1"$2"'));
  const allowed=new Set(['created','resumed','endpoint','status','cleanup','error','cleanup_error']);
  if(!event||typeof event!=='object'||Array.isArray(event)||typeof event.event!=='string'||!allowed.has(event.event)) fail('JSONL event schema invalid');
  return event;
}

function assertPhysicalPath(p, mustExist) {
  const resolved=path.resolve(p), root=path.parse(resolved).root;
  let cursor=root;
  for(const part of resolved.slice(root.length).split(path.sep).filter(Boolean)) {
    cursor=path.join(cursor,part);
    try { const st=fs.lstatSync(cursor); if(st.isSymbolicLink()) fail(`reparse/symlink ancestor: ${cursor}`); }
    catch(e) { if(e?.code!=='ENOENT') throw e; }
  }
  if(mustExist) {
    const real=fs.realpathSync.native(resolved);
    if(real.toLowerCase()!==resolved.toLowerCase()) fail(`physical path differs from lexical path: ${resolved} -> ${real}`);
  }
  return resolved;
}

}

function launchReceipt(child, completion, stdout, stderr, pending, events, parseErrors, streamErrors) {
  child.on('error',e=>streamErrors.push({stream:'child',error:String(e)}));
  for(const [name,s] of [['stdin',child.stdin],['stdout',child.stdout],['stderr',child.stderr]])
    s.on('error',e=>streamErrors.push({stream:name,error:String(e)}));
  child.stdout.setEncoding('utf8').on('data',chunk=>{stdout.value+=chunk;pending.value+=chunk;let n;while((n=pending.value.indexOf('\n'))>=0){const line=pending.value.slice(0,n);pending.value=pending.value.slice(n+1);try{events.push(parseEvent(line));}catch(e){parseErrors.push({line,error:String(e)});}}});
  child.stderr.setEncoding('utf8').on('data',chunk=>stderr.value+=chunk);
  child.once('close',(code,signal)=>completion.resolve({code,signal,closed:true}));
}
function waitUntil(p,ms) {
  let timer;
  return Promise.race([p,new Promise(resolve=>{timer=setTimeout(()=>resolve({timeout:true}),ms);})]).finally(()=>clearTimeout(timer));
}
function writeCommand(child,command) {
  return new Promise((resolve,reject)=>{
    if(child.stdin.destroyed||child.stdin.writableEnded||child.stdin.writableFinished) return reject(new Error('helper stdin already closed'));
    child.stdin.write(JSON.stringify({command})+'\n',err=>err?reject(err):resolve());
  });
}

async function runPipeline(cfg) {
  const pf=preflight(cfg),out=cfg.outDir; fs.mkdirSync(out,{recursive:false});
  const baseline={source:pf.sourceManifest,executable:{path:cfg.exe,sha256:digest(cfg.exe)},helper:{path:cfg.helper,sha256:digest(cfg.helper)},variant:{path:'Gear_blank.moproj',sha256:sha(pf.variant.bytes),offsets:pf.variant.offsets}};
  writeJsonExclusive(path.join(out,'preflight.json'),baseline); const completed=[];
  for(const c of cfg.cases) { if(cfg.beforeCase) await cfg.beforeCase(c); completed.push(await runCase(pf,cfg,c,out)); }
  const effects=completed.map(x=>({name:x.name,added:x.delta.added.map(e=>e.path),removed:x.delta.removed.map(e=>e.path),changed:x.delta.changed.map(e=>e.path)}));
  const hasLubricants=x=>x.added.some(p=>p.toLowerCase().split(/[\\/]/).includes('lubricants'));
  const controlHasLubricants=hasLubricants(effects[0]),variantHasLubricants=hasLubricants(effects[1]);
  const interpretation=!controlHasLubricants?'inconclusive_control_did_not_add_lubricants':variantHasLubricants?'variant_also_added_lubricants_positive_id_not_shown_necessary':'consistent_with_positive_id_path_but_not_causal_proof';
  const comparison={control:effects[0],negative_ids:effects[1],controlHasLubricants,variantHasLubricants,interpretation,
    sharedAdded:effects[0].added.filter(x=>effects[1].added.includes(x)),controlOnlyAdded:effects[0].added.filter(x=>!effects[1].added.includes(x)),negativeOnlyAdded:effects[1].added.filter(x=>!effects[0].added.includes(x))};
  writeJsonExclusive(path.join(out,'effect-comparison.json'),comparison); return {out,comparison};
}

  const cfg={exe:EXE,exeSha:EXE_SHA256,helper,helperSha:HELPER_SHA256,source:path.resolve(projectRoot,SOURCE),cases:CASES,strictShortPaths:true,pinSource:true,
    python:'C:\\Program Files\\Python312\\python.exe',watchdogSeconds:WATCHDOG_SECONDS,observeMs:OBSERVE_MS,
    expectedStructure:{control:'Die Stress,1,SfOprDiestress|Forming (2),2,SfOprForming',negative_ids:'Die Stress,-1,SfOprDiestress|Forming (2),-2,SfOprForming'},
    targetArgs:(root)=>['-m','open','-f',encodeQChar(path.join(root,'Gear_blank.moproj'))],outDir:path.join(projectRoot,'analysis',`resume217-plain-run-${new Date().toISOString().replaceAll(':','-')}`)};
  return runPipeline(cfg);
}
async function dryRun() {
  const sandbox=fs.mkdtempSync(path.join(projectRoot,'analysis','cp217-harmless-'));
  const source=path.join(sandbox,'source'),casesRoot=path.join(sandbox,'cases'),fixture=path.join(here,'tests','harmless_suspended_child.py');
  fs.mkdirSync(source);fs.mkdirSync(casesRoot);
  for(const d of ['Operations','Objects','Materials','Equipments']) fs.mkdirSync(path.join(source,d));
  const xml='<Root><Operation_List><Operation><Name>Die Stress</Name><Id>1</Id><Type>SfOprDiestress</Type></Operation><Operation><Name>Forming (2)</Name><Id>2</Id><Type>SfOprForming</Type></Operation></Operation_List></Root>';
  fs.writeFileSync(path.join(source,'Gear_blank.moproj'),xml);
  const cases=[{name:'control',root:path.join(casesRoot,'control')},{name:'negative_ids',root:path.join(casesRoot,'negative_ids')}];
  const python='C:\\Program Files\\Python312\\python.exe';
  const cfg={exe:python,exeSha:digest(python),helper,helperSha:HELPER_SHA256,source,cases,strictShortPaths:false,pinSource:false,python,watchdogSeconds:20,observeMs:100,
    expectedStructure:{control:'Die Stress,1,SfOprDiestress|Forming (2),2,SfOprForming',negative_ids:'Die Stress,-1,SfOprDiestress|Forming (2),-2,SfOprForming'},
    targetArgs:(root)=>[fixture,'-m','open','-f',encodeQChar(path.join(root,'Gear_blank.moproj'))],
    outDir:path.join(sandbox,'receipts')};
  const result=await runPipeline(cfg);
  const raceRoot=path.join(casesRoot,'race-existing'),raceOut=path.join(sandbox,'receipts-race');
  let raceRejected=false;
  try { await runPipeline({...cfg,cases:[{name:'control',root:raceRoot}],outDir:raceOut,beforeCase:()=>fs.mkdirSync(raceRoot)}); }
  catch { raceRejected=fs.existsSync(path.join(raceOut,'control','clone-error.json'))&&fs.readdirSync(raceRoot).length===0; }
  if(!raceRejected) fail('race-created destination fixture was not refused cleanly');
  const missingRoot=path.join(casesRoot,'inventory-failure'),missingOut=path.join(sandbox,'receipts-inventory-failure');
  let inventoryFailureCaptured=false;
  try { await runPipeline({...cfg,cases:[{name:'control',root:missingRoot}],outDir:missingOut,afterCleanup:async root=>{fs.rmSync(root,{recursive:true,force:true});}}); }
  catch {
    const resultPath=path.join(missingOut,'control','result.json');
    if(fs.existsSync(resultPath)) { const r=JSON.parse(fs.readFileSync(resultPath,'utf8')); inventoryFailureCaptured=!r.complete&&typeof r.after?.error==='string'&&fs.existsSync(path.join(missingOut,'control','helper.stdout.raw')); }
  }
  if(!inventoryFailureCaptured) fail('after-inventory failure fixture was not retained/rejected');
  process.stdout.write(JSON.stringify({dryRun:'passed',originalLaunch:false,sandbox,enabledRun:result,raceCreatedRootRejected:raceRejected,afterInventoryFailureCaptured:inventoryFailureCaptured})+'\n');
}
if(process.argv[1] && path.resolve(process.argv[1])===fileURLToPath(import.meta.url)) {
  try { if(process.argv[2]==='--self-test') selfTest().catch(e=>{process.stderr.write(String(e.stack||e)+'\n');process.exitCode=1;}); else if(process.argv[2]==='--dry-run-harmless') dryRun().catch(e=>{process.stderr.write(String(e.stack||e)+'\n');process.exitCode=3;}); else run().then(x=>process.stdout.write(JSON.stringify(x)+'\n')).catch(e=>{process.stderr.write(String(e.stack||e)+'\n');process.exitCode=2;}); }
  catch(e) { process.stderr.write(String(e.stack||e)+'\n'); process.exitCode=1; }
}
